```python
from decimal import Decimal
from collections import defaultdict

from django.db import transaction
from django.shortcuts import get_object_or_404
from django.utils import timezone
from rest_framework.exceptions import PermissionDenied, ValidationError

from core.api import Conflict
from core.audit import audit
from core.models import Portafolio, Posicion, IndiceRiesgo
from .numbers import quantize


def check_writer(user, portfolio=None):
    if user.rol not in ['ADMIN', 'ANALISTA']:
        raise PermissionDenied('Tu rol no permite modificar portafolios.')
    if portfolio and user.rol != 'ADMIN' and portfolio.usuario_id != user.pk:
        raise PermissionDenied('Solo el dueño o ADMIN puede modificar este portafolio.')


def lock_portfolio(user, pk, version):
    portfolio = get_object_or_404(Portafolio.objects.select_for_update(), pk=pk)
    check_writer(user, portfolio)
    if version is None:
        raise ValidationError({'version': 'Envía la versión actual del portafolio.'})
    if portfolio.version != version:
        raise Conflict()
    return portfolio


def touch(portfolio):
    portfolio.version += 1
    portfolio.fecha_modificacion = timezone.now()
    portfolio.save()


@transaction.atomic
def create_portfolio(user, data):
    check_writer(user)
    portfolio = Portafolio(usuario=user, **data)
    portfolio.full_clean()
    portfolio.save()
    audit(user, 'CREAR', portfolio)
    return portfolio


@transaction.atomic
def update_portfolio(user, pk, version, data):
    portfolio = lock_portfolio(user, pk, version)
    for key, value in data.items():
        setattr(portfolio, key, value)
    portfolio.full_clean()
    touch(portfolio)
    audit(user, 'EDITAR', portfolio, {'version': portfolio.version})
    return portfolio


@transaction.atomic
def delete_portfolio(user, pk, version):
    portfolio = lock_portfolio(user, pk, version)
    portfolio.activo = False
    touch(portfolio)
    audit(user, 'ELIMINAR', portfolio)


def validate_position(position, exclude_pk=None):
    position.full_clean()
    if position.tipo_activo == 'MONEDA' and position.pais.moneda_id == 'USD':
        raise ValidationError({'tipo_activo': 'No se permite MONEDA para un país cuya moneda es USD.'})
    active = Posicion.objects.filter(portafolio=position.portafolio, fecha_salida__isnull=True)
    if exclude_pk:
        active = active.exclude(pk=exclude_pk)
    if active.filter(pais=position.pais, tipo_activo=position.tipo_activo).count() >= 2:
        raise ValidationError({'tipo_activo': 'Máximo dos posiciones activas del mismo tipo en el mismo país.'})
    total = sum(active.values_list('monto_inversion_usd', flat=True), Decimal('0')) + position.monto_inversion_usd
    if total > Decimal('50000000'):
        raise ValidationError({'monto_inversion_usd': 'El total activo no puede superar USD 50.000.000.'})


@transaction.atomic
def add_position(user, pk, version, data):
    portfolio = lock_portfolio(user, pk, version)
    position = Posicion(portafolio=portfolio, **data)
    validate_position(position)
    position.save()
    touch(portfolio)
    audit(user, 'CREAR', position, {'portafolio_id': portfolio.pk, 'version': portfolio.version})
    return position


@transaction.atomic
def edit_position(user, pk, position_pk, version, data):
    portfolio = lock_portfolio(user, pk, version)
    position = get_object_or_404(Posicion.objects, pk=position_pk, portafolio=portfolio)
    if position.fecha_salida is not None:
        raise Conflict('La posición ya está cerrada y conserva su histórico.')
    for key, value in data.items():
        setattr(position, key, value)
    validate_position(position, exclude_pk=position.pk)
    position.save()
    touch(portfolio)
    audit(user, 'EDITAR', position, {'version': portfolio.version})
    return position


@transaction.atomic
def close_position(user, pk, position_pk, version):
    portfolio = lock_portfolio(user, pk, version)
    position = get_object_or_404(Posicion.objects, pk=position_pk, portafolio=portfolio)
    if position.fecha_salida is not None:
        raise Conflict('La posición ya está cerrada.')
    position.fecha_salida = timezone.now().date()
    position.full_clean()
    position.save(update_fields=['fecha_salida'])
    touch(portfolio)
    audit(user, 'ELIMINAR', position, {'cierre': True, 'version': portfolio.version})
    return position


def portfolio_summary(portfolio):
    by_country, by_type = defaultdict(lambda: Decimal('0')), defaultdict(lambda: Decimal('0'))
    total, covered, weighted = Decimal('0'), Decimal('0'), Decimal('0')
    missing = set()
    risks = {}
    positions = list(portfolio.posiciones.filter(fecha_salida__isnull=True).select_related('pais'))
    for position in positions:
        amount = position.monto_inversion_usd
        total += amount
        by_country[position.pais_id] += amount
        by_type[position.tipo_activo] += amount
        if position.pais_id not in risks:
            risks[position.pais_id] = IndiceRiesgo.objects.filter(pais=position.pais).first()
        risk = risks[position.pais_id]
        if risk:
            weighted += risk.indice_compuesto * amount
            covered += amount
            if risk.detalle_calculo.get('parcial'):
                missing.add(position.pais_id)
        else:
            missing.add(position.pais_id)
    def distribution(groups):
        return [{'codigo': key, 'monto_usd': str(quantize(value)),
                 'porcentaje': str(quantize(value / total * 100)) if total else '0.00'} for key, value in sorted(groups.items())]
    return {'total_invertido_usd': str(quantize(total)), 'posiciones_activas': len(positions),
        'por_pais': distribution(by_country), 'por_tipo_activo': distribution(by_type),
        'riesgo_promedio_ponderado': str(quantize(weighted / covered)) if covered else None,
        'monto_con_riesgo_usd': str(quantize(covered)), 'paises_con_datos_faltantes': sorted(missing),
        'version': portfolio.version}
```

```python
from django.db.models import Q
from django.http import HttpResponse
from rest_framework import serializers, viewsets, status
from rest_framework.decorators import action
from rest_framework.response import Response
from drf_spectacular.utils import extend_schema, extend_schema_field, OpenApiParameter, OpenApiTypes

from .models import Portafolio, Posicion
from .permissions import PuedeEditarPortafolio
from .services import portfolios as domain
from .services.pdf_report import portfolio_pdf
from .audit import audit


class PositionSerializer(serializers.ModelSerializer):
    class Meta:
        model = Posicion
        fields = ['id', 'portafolio', 'pais', 'tipo_activo', 'monto_inversion_usd', 'fecha_entrada', 'fecha_salida', 'notas']
        read_only_fields = ['id', 'portafolio', 'fecha_salida']

    def validate_fecha_entrada(self, value):
        from django.utils import timezone
        if value > timezone.now().date():
            raise serializers.ValidationError('La entrada no puede ser futura.')
        return value


class PositionWriteSerializer(PositionSerializer):
    version = serializers.IntegerField(min_value=1, write_only=True)

    class Meta(PositionSerializer.Meta):
        fields = PositionSerializer.Meta.fields + ['version']


class VersionSerializer(serializers.Serializer):
    version = serializers.IntegerField(min_value=1)


class DistributionSerializer(serializers.Serializer):
    codigo = serializers.CharField()
    monto_usd = serializers.DecimalField(max_digits=16, decimal_places=2)
    porcentaje = serializers.DecimalField(max_digits=7, decimal_places=2)


class PortfolioSummarySerializer(serializers.Serializer):
    total_invertido_usd = serializers.DecimalField(max_digits=16, decimal_places=2)
    posiciones_activas = serializers.IntegerField()
    por_pais = DistributionSerializer(many=True)
    por_tipo_activo = DistributionSerializer(many=True)
    riesgo_promedio_ponderado = serializers.DecimalField(max_digits=5, decimal_places=2, allow_null=True)
    monto_con_riesgo_usd = serializers.DecimalField(max_digits=16, decimal_places=2)
    paises_con_datos_faltantes = serializers.ListField(child=serializers.CharField())
    version = serializers.IntegerField()


class PortfolioSerializer(serializers.ModelSerializer):
    version = serializers.IntegerField(min_value=1, required=False)

    class Meta:
        model = Portafolio
        fields = ['id', 'nombre', 'descripcion', 'usuario', 'es_publico', 'activo', 'fecha_creacion', 'fecha_modificacion', 'version']
        read_only_fields = ['id', 'usuario', 'activo', 'fecha_creacion', 'fecha_modificacion']
        validators = []

    def validate_nombre(self, value):
        owner = self.instance.usuario if self.instance else self.context['request'].user
        query = Portafolio.todos.filter(usuario=owner, nombre=value)
        if self.instance:
            query = query.exclude(pk=self.instance.pk)
        if query.exists():
            raise serializers.ValidationError('Ya existe un portafolio con este nombre para el usuario.')
        return value

    def validate(self, attrs):
        if self.instance and 'version' not in attrs:
            raise serializers.ValidationError({'version': 'Envía la versión actual del portafolio.'})
        return attrs


class PortfolioDetailSerializer(PortfolioSerializer):
    posiciones = PositionSerializer(many=True, read_only=True)
    metricas = serializers.SerializerMethodField()

    class Meta(PortfolioSerializer.Meta):
        fields = PortfolioSerializer.Meta.fields + ['posiciones', 'metricas']

    @extend_schema_field(PortfolioSummarySerializer)
    def get_metricas(self, obj):
        return domain.portfolio_summary(obj)


class AvailabilitySerializer(serializers.Serializer):
    disponible = serializers.BooleanField()


class AvailabilityQuerySerializer(serializers.Serializer):
    nombre = serializers.CharField(min_length=3, max_length=100)
    excluir_id = serializers.IntegerField(min_value=1, required=False)


class PortafolioViewSet(viewsets.ModelViewSet):
    queryset = Portafolio.objects.all()
    serializer_class = PortfolioSerializer
    permission_classes = [PuedeEditarPortafolio]
    http_method_names = ['get', 'post', 'put', 'delete', 'head', 'options']
    search_fields = ['nombre', 'descripcion']
    ordering_fields = ['nombre', 'fecha_creacion', 'fecha_modificacion']
    filterset_fields = ['es_publico']

    def get_queryset(self):
        if getattr(self, 'swagger_fake_view', False):
            return Portafolio.objects.none()
        records = Portafolio.objects.select_related('usuario').all()
        if self.request.user.rol != 'ADMIN':
            records = records.filter(Q(usuario=self.request.user) | Q(es_publico=True))
        return records

    def get_serializer_class(self):
        return PortfolioDetailSerializer if self.action == 'retrieve' else super().get_serializer_class()

    def create(self, request, *args, **kwargs):
        serializer = self.get_serializer(data=request.data)
        serializer.is_valid(raise_exception=True)
        data = dict(serializer.validated_data)
        data.pop('version', None)
        portfolio = domain.create_portfolio(request.user, data)
        return Response(self.get_serializer(portfolio).data, status=status.HTTP_201_CREATED)

    def update(self, request, *args, **kwargs):
        portfolio = self.get_object()
        serializer = self.get_serializer(portfolio, data=request.data)
        serializer.is_valid(raise_exception=True)
        data = dict(serializer.validated_data)
        version = data.pop('version')
        portfolio = domain.update_portfolio(request.user, portfolio.pk, version, data)
        return Response(self.get_serializer(portfolio).data)

    def destroy(self, request, *args, **kwargs):
        portfolio = self.get_object()
        serializer = VersionSerializer(data=request.data)
        serializer.is_valid(raise_exception=True)
        domain.delete_portfolio(request.user, portfolio.pk, serializer.validated_data['version'])
        return Response(status=204)

    @extend_schema(request=PositionWriteSerializer, responses={201: PositionSerializer})
    @action(detail=True, methods=['post'])
    def posiciones(self, request, pk=None):
        portfolio = self.get_object()
        serializer = PositionWriteSerializer(data=request.data)
        serializer.is_valid(raise_exception=True)
        data = dict(serializer.validated_data)
        version = data.pop('version')
        result = domain.add_position(request.user, portfolio.pk, version, data)
        return Response(PositionSerializer(result).data, status=201)

    @extend_schema(methods=['PUT'], request=PositionWriteSerializer, responses=PositionSerializer,
                   parameters=[OpenApiParameter('posicion_id', int, OpenApiParameter.PATH)])
    @extend_schema(methods=['DELETE'], request=VersionSerializer, responses=PositionSerializer,
                   parameters=[OpenApiParameter('posicion_id', int, OpenApiParameter.PATH)])
    @action(detail=True, methods=['put', 'delete'], url_path=r'posiciones/(?P<posicion_id>\d+)')
    def posicion(self, request, pk=None, posicion_id=None):
        portfolio = self.get_object()
        if request.method == 'DELETE':
            serializer = VersionSerializer(data=request.data)
            serializer.is_valid(raise_exception=True)
            result = domain.close_position(request.user, portfolio.pk, posicion_id, serializer.validated_data['version'])
        else:
            serializer = PositionWriteSerializer(data=request.data)
            serializer.is_valid(raise_exception=True)
            data = dict(serializer.validated_data)
            version = data.pop('version')
            result = domain.edit_position(request.user, portfolio.pk, posicion_id, version, data)
        return Response(PositionSerializer(result).data)

    @extend_schema(responses=PortfolioSummarySerializer)
    @action(detail=True, methods=['get'])
    def resumen(self, request, pk=None):
        return Response(domain.portfolio_summary(self.get_object()))

    @extend_schema(responses={(200, 'application/pdf'): OpenApiTypes.BINARY})
    @action(detail=True, methods=['get'], url_path='export/pdf')
    def export_pdf(self, request, pk=None):
        portfolio = self.get_object()
        content = portfolio_pdf(portfolio)
        audit(request.user, 'EXPORT', portfolio)
        response = HttpResponse(content, content_type='application/pdf')
        response['Content-Disposition'] = f'attachment; filename="portafolio-{portfolio.pk}.pdf"'
        return response

    @extend_schema(parameters=[AvailabilityQuerySerializer], responses=AvailabilitySerializer)
    @action(detail=False, methods=['get'], url_path='nombre-disponible')
    def nombre_disponible(self, request):
        serializer = AvailabilityQuerySerializer(data=request.query_params)
        serializer.is_valid(raise_exception=True)
        records = Portafolio.todos.filter(usuario=request.user, nombre=serializer.validated_data['nombre'])
        if 'excluir_id' in serializer.validated_data:
            records = records.exclude(pk=serializer.validated_data['excluir_id'])
        return Response({'disponible': not records.exists()})
```

## backend/core/services/pdf_report.py

```python
from io import BytesIO
from xml.sax.saxutils import escape
from reportlab.lib import colors
from reportlab.lib.styles import getSampleStyleSheet
from reportlab.platypus import SimpleDocTemplate, Paragraph, Spacer, Table, TableStyle
from .portfolios import portfolio_summary


def portfolio_pdf(portfolio):
    output = BytesIO()
    styles = getSampleStyleSheet()
    doc = SimpleDocTemplate(output, title='Reporte de portafolio DataPulse Latam')
    summary = portfolio_summary(portfolio)
    story = [Paragraph('DataPulse Latam', styles['Title']),
        Paragraph(escape(portfolio.nombre), styles['Heading1']),
        Paragraph(escape(portfolio.descripcion or 'Sin descripción'), styles['BodyText']), Spacer(1, 12),
        Paragraph(f'Total activo USD: {summary["total_invertido_usd"]}', styles['BodyText']),
        Paragraph(f'IRPC ponderado: {summary["riesgo_promedio_ponderado"] or "Sin datos"}', styles['BodyText']),
        Paragraph('Un score más alto indica menor riesgo. Los cálculos parciales se indican en el detalle.', styles['BodyText']),
        Spacer(1, 12)]
    rows = [['País', 'Tipo', 'Monto USD', 'Entrada', 'Salida']]
    for position in portfolio.posiciones.all():
        rows.append([position.pais_id, position.tipo_activo, str(position.monto_inversion_usd),
                     str(position.fecha_entrada), str(position.fecha_salida or 'Activa')])
    table = Table(rows, repeatRows=1)
    table.setStyle(TableStyle([('BACKGROUND', (0, 0), (-1, 0), colors.HexColor('#173b58')),
        ('TEXTCOLOR', (0, 0), (-1, 0), colors.white), ('FONTSIZE', (0, 0), (-1, -1), 8),
        ('BOTTOMPADDING', (0, 0), (-1, 0), 10), ('GRID', (0, 0), (-1, -1), 1, colors.lightgrey)]))
    story.append(table)
    if summary['paises_con_datos_faltantes']:
        story.append(Paragraph('Datos de riesgo incompletos: ' + ', '.join(summary['paises_con_datos_faltantes']), styles['BodyText']))
    doc.build(story)
    return output.getvalue()
```
