import logging

from django.core.exceptions import ValidationError as DjangoValidationError
from django.db import IntegrityError
from rest_framework.exceptions import APIException, ValidationError
from rest_framework.pagination import PageNumberPagination
from rest_framework.response import Response
from rest_framework.views import exception_handler as drf_exception_handler

logger = logging.getLogger('datapulse')


class Conflict(APIException):
    status_code = 409
    default_detail = 'El registro cambió. Actualiza los datos antes de reintentar.'
    default_code = 'conflict'


class Pagination(PageNumberPagination):
    page_size = 20
    page_size_query_param = 'page_size'
    max_page_size = 500


def exception_handler(exc, context):
    if isinstance(exc, DjangoValidationError):
        exc = ValidationError(exc.message_dict if hasattr(exc, 'message_dict') else exc.messages)
    if isinstance(exc, IntegrityError):
        exc = Conflict('La operación entra en conflicto con un registro existente.')
    result = drf_exception_handler(exc, context)
    if result is None:
        logger.error('error_interno', exc_info=True)
        return Response({'error': {'code': 'internal_error', 'message': 'Error interno del servidor.', 'details': {}}}, status=500)
    messages = {
        400: 'Revisa los datos enviados.', 401: 'Autenticación requerida o token inválido.',
        403: 'No tienes permisos para esta acción.', 404: 'No se encontró el recurso.',
        405: 'Método no permitido.', 409: 'Conflicto al actualizar los datos.',
        429: 'Demasiadas solicitudes. Intenta más tarde.',
    }
    code = getattr(exc, 'default_code', 'error')
    detail = result.data
    message = str(detail['detail']) if result.status_code == 409 and 'detail' in detail else messages.get(result.status_code, 'No se pudo completar la operación.')
    result.data = {'error': {'code': code, 'message': message, 'details': detail}}
    return result
